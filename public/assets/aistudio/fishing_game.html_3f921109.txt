// 精準計算任意 (x, z, time) 座標的水面波浪高度
function getWaterHeight(x, z, time) {
    const w1 = Math.sin(x * 0.25 + time * 1.8) * Math.cos(z * 0.2 + time * 1.4) * 0.12;
    const w2 = Math.sin((x * 0.6 - z * 0.4) + time * 2.6) * 0.04;
    const w3 = Math.cos(z * 0.8 + time * 3.2) * 0.02;
    return w1 + w2 + w3;
}

// 更新水面頂點
function updateWaterGeometry(time) {
    const pos = waterGeo.attributes.position;
    for (let i = 0; i < pos.count; i++) {
        const vx = pos.getX(i);
        const vz = pos.getZ(i);
        // 僅對視野核心區域進行高頻率波浪起伏
        pos.setY(i, getWaterHeight(vx, vz, time));
    }
    pos.needsUpdate = true;
    waterGeo.computeVertexNormals();
}
